```python
class FuncTestCase(Base):
    __tablename__ = "func_test_cases"

    id               = Column(Integer, primary_key=True)
    project_id       = Column(Integer, ForeignKey("projects.id", ondelete="CASCADE"))
    group_id         = Column(Integer, ForeignKey("func_case_groups.id"), nullable=True)
    title            = Column(String(200), nullable=False)
    precondition     = Column(Text, nullable=True)
    priority         = Column(String(10), default="P2")   # P0/P1/P2/P3
    status           = Column(String(20), default="draft") # draft/pending_review/approved/deprecated
    steps            = Column(JSONB, default=[])           # [{step, expected}]
    tags             = Column(ARRAY(String), default=[])
    estimated_hours  = Column(Float, nullable=True)
    created_by       = Column(Integer, ForeignKey("users.id"), nullable=True)
    created_at       = Column(DateTime, default=datetime.utcnow)
    updated_at       = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

    group = relationship("FuncCaseGroup", back_populates="cases")
```

#### XMind 导入核心逻辑（伪代码）

```python
# backend/app/services/xmind_service.py

def import_xmind(file_bytes, project_id, db):
    data = xmindparser.load(file_bytes)   # 解析为 Python dict 树
    root = data[0]["topic"]               # 取第一个 sheet 的根节点

    def process_node(node, parent_group_id=None):
        children = node.get("topics", [])
        # 有子节点 → 视为分组
        if children:
            group = create_func_case_group(name=node["title"], parent_id=parent_group_id)
            for child in children:
                process_node(child, parent_group_id=group.id)
        # 叶子节点 → 视为用例
        else:
            create_func_test_case(title=node["title"], group_id=parent_group_id)

    process_node(root)
```

#### XMind 导出核心逻辑（伪代码）

```python
def export_xmind(project_id, db):
    tree = get_case_tree(project_id, db)   # 拿到完整树

    def build_node(item):
        node = {"title": item["name"], "topics": []}
        for child in item.get("children", []):
            node["topics"].append(build_node(child))
        for case in item.get("cases", []):
            node["topics"].append({"title": case["title"]})
        return node

    content = [{"title": "测试用例", "topic": build_node(tree)}]

    # 打包为 .xmind (ZIP)
    buf = BytesIO()
    with zipfile.ZipFile(buf, "w") as zf:
        zf.writestr("content.json", json.dumps(content))
    return buf.getvalue()
```
